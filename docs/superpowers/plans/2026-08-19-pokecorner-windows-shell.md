# PokéCorner Windows Shell Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build `PokeCorner.App` — the WPF (.NET 8) desktop shell that turns `PokeCorner.Core` (see `docs/superpowers/plans/2026-08-19-pokecorner-core-domain.md`, which must be implemented first) into the actual running desktop pet: tray icon, floating overlay sprite, global activity hooks, fullscreen avoidance, collection/settings popups, and first-run onboarding.

**Architecture:** One WPF project (`src/PokeCorner.App`) referencing `PokeCorner.Core`. Every piece of logic that *can* be expressed without touching a live window handle, a message pump, or a P/Invoke call is pulled out into a small, pure, unit-tested class (`InputEventCounter`, `FrameSequencer`, `FullscreenDetector`, `CollectionViewModel`, `SettingsViewModel`, `FirstRunFlow`) living in `tests/PokeCorner.App.Tests`. The remaining thin glue (`GlobalInputHook`, `SpriteAnimator`, `ForegroundWindowMonitor`, `OverlayWindow`, `TrayIconService`, `AutostartRegistry`) only wires that pure logic to real Win32/WPF APIs, and is verified with the manual smoke-test checklists embedded in each task (per the spec's own Testing section: UI/hook behavior isn't cheaply automatable, so each task ends with concrete manual steps instead of skipping verification).

**Tech Stack:** .NET 8 (`net8.0-windows`), WPF, Windows Forms interop for the tray icon (`System.Windows.Forms.NotifyIcon` via `UseWindowsForms`), Win32 P/Invoke (`user32.dll`), xUnit.

## Global Constraints

- Never store, log, or persist raw keystroke content or window titles — `GlobalInputHook` must only ever call `InputEventCounter.RecordEvent()`, never read or forward the key/mouse data it intercepts (from the spec's activity-tracking privacy requirement, also enforced structurally in the Core plan's `ActivityAccumulator`).
- The pet must not obscure fullscreen apps (games, video, presentations) — `ForegroundWindowMonitor` + `FullscreenDetector` must auto-hide the overlay whenever the foreground window is fullscreen, and restore it when focus moves away (spec: "Не мешать пользователю").
- Token-usage tracking stays optional and off by default — the composition root only starts `ClaudeCodeUsageProvider` when `AppSettings.TokenTrackingEnabled` is true.
- All data stays local under `%AppData%\PokeCorner` — no cloud sync, no accounts (from the spec's "Данные и edge cases" section).
- Activity tracking has a visible on/off master switch with a tray indicator, and Hardcore care-mode neglect never deletes a caught individual — only the explicit "отпустить" action does (spec: "Уход за питомцем", "Поимка новых покемонов и коллекция").

```csharp
namespace PokeCorner.App.Input;

public sealed class InputEventCounter
{
    private long _pendingUnits;

    public void RecordEvent() => Interlocked.Increment(ref _pendingUnits);

    public long DrainUnits() => Interlocked.Exchange(ref _pendingUnits, 0);
}
```

```csharp
// src/PokeCorner.App/Input/GlobalInputHook.cs
using System.Runtime.InteropServices;

namespace PokeCorner.App.Input;

public sealed class GlobalInputHook : IDisposable
{
    private const int WH_MOUSE_LL = 14;
    private const int WH_KEYBOARD_LL = 13;

    private readonly InputEventCounter _counter;
    private readonly LowLevelProc _mouseProc;
    private readonly LowLevelProc _keyboardProc;
    private IntPtr _mouseHookHandle;
    private IntPtr _keyboardHookHandle;

    private delegate IntPtr LowLevelProc(int nCode, IntPtr wParam, IntPtr lParam);

    public GlobalInputHook(InputEventCounter counter)
    {
        _counter = counter;
        // Keep delegate instances alive for the hook's lifetime — the CLR would
        // otherwise be free to collect them since SetWindowsHookEx only holds a
        // native function pointer, not a managed reference.
        _mouseProc = MouseHookCallback;
        _keyboardProc = KeyboardHookCallback;
    }

    public void Start()
    {
        using var currentModule = System.Diagnostics.Process.GetCurrentProcess().MainModule!;
        var moduleHandle = GetModuleHandle(currentModule.ModuleName);

        _mouseHookHandle = SetWindowsHookEx(WH_MOUSE_LL, _mouseProc, moduleHandle, 0);
        _keyboardHookHandle = SetWindowsHookEx(WH_KEYBOARD_LL, _keyboardProc, moduleHandle, 0);
    }

    public void Stop()
    {
        if (_mouseHookHandle != IntPtr.Zero) UnhookWindowsHookEx(_mouseHookHandle);
        if (_keyboardHookHandle != IntPtr.Zero) UnhookWindowsHookEx(_keyboardHookHandle);
        _mouseHookHandle = IntPtr.Zero;
        _keyboardHookHandle = IntPtr.Zero;
    }

    public void Dispose() => Stop();

    private IntPtr MouseHookCallback(int nCode, IntPtr wParam, IntPtr lParam)
    {
        if (nCode >= 0) _counter.RecordEvent(); // count only — never inspect wParam/lParam contents.
        return CallNextHookEx(_mouseHookHandle, nCode, wParam, lParam);
    }

    private IntPtr KeyboardHookCallback(int nCode, IntPtr wParam, IntPtr lParam)
    {
        if (nCode >= 0) _counter.RecordEvent(); // count only — never read the key code.
        return CallNextHookEx(_keyboardHookHandle, nCode, wParam, lParam);
    }

    [DllImport("user32.dll", SetLastError = true)]
    private static extern IntPtr SetWindowsHookEx(int idHook, LowLevelProc lpfn, IntPtr hMod, uint dwThreadId);

    [DllImport("user32.dll", SetLastError = true)]
    [return: MarshalAs(UnmanagedType.Bool)]
    private static extern bool UnhookWindowsHookEx(IntPtr hhk);

    [DllImport("user32.dll")]
    private static extern IntPtr CallNextHookEx(IntPtr hhk, int nCode, IntPtr wParam, IntPtr lParam);

    [DllImport("kernel32.dll", CharSet = CharSet.Auto, SetLastError = true)]
    private static extern IntPtr GetModuleHandle(string lpModuleName);
}
```

```csharp
// src/PokeCorner.App/Overlay/FrameSequencer.cs
namespace PokeCorner.App.Overlay;

public sealed class FrameSequencer
{
    private readonly int _frameCount;
    private readonly TimeSpan _frameDuration;

    public FrameSequencer(int frameCount, TimeSpan frameDuration)
    {
        if (frameCount <= 0) throw new ArgumentOutOfRangeException(nameof(frameCount));
        _frameCount = frameCount;
        _frameDuration = frameDuration;
    }

    public int FrameIndexAt(TimeSpan elapsed)
    {
        var cycleDurationMs = _frameDuration.TotalMilliseconds * _frameCount;
        var wrappedMs = elapsed.TotalMilliseconds % cycleDurationMs;
        return (int)(wrappedMs / _frameDuration.TotalMilliseconds);
    }
}
```

```csharp
using System.Windows.Controls;
using System.Windows.Media;
using System.Windows.Threading;

namespace PokeCorner.App.Overlay;

public sealed class SpriteAnimator
{
    private readonly Image _target;
    private readonly FrameSequencer _sequencer;
    private readonly IReadOnlyList<ImageSource> _frames;
    private readonly DispatcherTimer _timer;
    private readonly System.Diagnostics.Stopwatch _stopwatch = new();

    public SpriteAnimator(Image target, FrameSequencer sequencer, IReadOnlyList<ImageSource> frames)
    {
        _target = target;
        _sequencer = sequencer;
        _frames = frames;
        _timer = new DispatcherTimer(DispatcherPriority.Render) { Interval = TimeSpan.FromMilliseconds(33) };
        _timer.Tick += (_, _) => Tick();
    }

    public void Start()
    {
        _stopwatch.Restart();
        _timer.Start();
    }

    public void Stop()
    {
        _timer.Stop();
        _stopwatch.Stop();
    }

    private void Tick()
    {
        var frameIndex = _sequencer.FrameIndexAt(_stopwatch.Elapsed);
        _target.Source = _frames[frameIndex];
    }
}
```

```csharp
// src/PokeCorner.App/Fullscreen/FullscreenDetector.cs
namespace PokeCorner.App.Fullscreen;

public readonly record struct ScreenRect(double X, double Y, double Width, double Height);

public static class FullscreenDetector
{
    private const double ToleranceScreenUnits = 0.5;

    public static bool IsFullscreen(ScreenRect windowRect, ScreenRect monitorRect, bool windowHasBorder)
    {
        if (windowHasBorder) return false;

        return Math.Abs(windowRect.X - monitorRect.X) < ToleranceScreenUnits
            && Math.Abs(windowRect.Y - monitorRect.Y) < ToleranceScreenUnits
            && Math.Abs(windowRect.Width - monitorRect.Width) < ToleranceScreenUnits
            && Math.Abs(windowRect.Height - monitorRect.Height) < ToleranceScreenUnits;
    }
}
```

```csharp
// src/PokeCorner.App/Fullscreen/ForegroundWindowMonitor.cs
using System.Runtime.InteropServices;
using System.Windows.Threading;

namespace PokeCorner.App.Fullscreen;

public sealed class ForegroundWindowMonitor : IDisposable
{
    private readonly Action _onFullscreenEntered;
    private readonly Action _onFullscreenExited;
    private readonly DispatcherTimer _pollTimer;
    private bool _wasFullscreen;

    public ForegroundWindowMonitor(Action onFullscreenEntered, Action onFullscreenExited)
    {
        _onFullscreenEntered = onFullscreenEntered;
        _onFullscreenExited = onFullscreenExited;
        _pollTimer = new DispatcherTimer { Interval = TimeSpan.FromMilliseconds(500) };
        _pollTimer.Tick += (_, _) => Poll();
    }

    public void Start() => _pollTimer.Start();
    public void Stop() => _pollTimer.Stop();
    public void Dispose() => Stop();

    private void Poll()
    {
        var isFullscreen = IsForegroundWindowFullscreen();
        if (isFullscreen && !_wasFullscreen) _onFullscreenEntered();
        else if (!isFullscreen && _wasFullscreen) _onFullscreenExited();
        _wasFullscreen = isFullscreen;
    }

    private static bool IsForegroundWindowFullscreen()
    {
        var foreground = GetForegroundWindow();
        if (foreground == IntPtr.Zero) return false;

        // The shell's own desktop/taskbar windows report as borderless full-monitor
        // too; exclude them by class name so the pet never hides on an empty desktop.
        var className = GetClassName(foreground);
        if (className is "Progman" or "WorkerW" or "Shell_TrayWnd") return false;

        if (!GetWindowRect(foreground, out var windowRect)) return false;

        var monitorHandle = MonitorFromWindow(foreground, MONITOR_DEFAULTTONEAREST);
        var monitorInfo = MONITORINFO.Create();
        if (!GetMonitorInfo(monitorHandle, ref monitorInfo)) return false;

        var style = GetWindowLong(foreground, GWL_STYLE);
        var hasBorder = (style & (WS_CAPTION | WS_THICKFRAME)) != 0;

        var windowScreenRect = new ScreenRect(windowRect.Left, windowRect.Top, windowRect.Right - windowRect.Left, windowRect.Bottom - windowRect.Top);
        var monitorScreenRect = new ScreenRect(
            monitorInfo.rcMonitor.Left, monitorInfo.rcMonitor.Top,
            monitorInfo.rcMonitor.Right - monitorInfo.rcMonitor.Left,
            monitorInfo.rcMonitor.Bottom - monitorInfo.rcMonitor.Top);

        return FullscreenDetector.IsFullscreen(windowScreenRect, monitorScreenRect, hasBorder);
    }

    private static string GetClassName(IntPtr hWnd)
    {
        var buffer = new System.Text.StringBuilder(256);
        GetClassName(hWnd, buffer, buffer.Capacity);
        return buffer.ToString();
    }

    private const int GWL_STYLE = -16;
    private const long WS_CAPTION = 0x00C00000L;
    private const long WS_THICKFRAME = 0x00040000L;
    private const uint MONITOR_DEFAULTTONEAREST = 2;

    [StructLayout(LayoutKind.Sequential)]
    private struct RECT { public int Left, Top, Right, Bottom; }

    [StructLayout(LayoutKind.Sequential)]
    private struct MONITORINFO
    {
        public uint cbSize;
        public RECT rcMonitor;
        public RECT rcWork;
        public uint dwFlags;

        public static MONITORINFO Create() => new() { cbSize = (uint)Marshal.SizeOf<MONITORINFO>() };
    }

    [DllImport("user32.dll")] private static extern IntPtr GetForegroundWindow();
    [DllImport("user32.dll")] private static extern bool GetWindowRect(IntPtr hWnd, out RECT rect);
    [DllImport("user32.dll")] private static extern IntPtr MonitorFromWindow(IntPtr hWnd, uint dwFlags);
    [DllImport("user32.dll", CharSet = CharSet.Auto)] private static extern bool GetMonitorInfo(IntPtr hMonitor, ref MONITORINFO lpmi);
    [DllImport("user32.dll")] private static extern long GetWindowLong(IntPtr hWnd, int nIndex);
    [DllImport("user32.dll", CharSet = CharSet.Auto)] private static extern int GetClassName(IntPtr hWnd, System.Text.StringBuilder lpClassName, int nMaxCount);
}
```

```csharp
    private readonly Icon _activeIcon;
    private readonly Icon _inactiveIcon;

    public event Action? OpenSettingsRequested;

        _activeIcon = SystemIcons.Application;
        _inactiveIcon = SystemIcons.Application;
        // Task 9 replaces both icons with the app's real tray art (a plain sprite icon,

        menu.Items.Add("Настройки", null, (_, _) => OpenSettingsRequested?.Invoke());
        menu.Items.Add(new ToolStripSeparator());
        menu.Items.Add("Выход", null, (_, _) => ExitRequested?.Invoke());

        _notifyIcon = new NotifyIcon
        {
            Icon = _activeIcon,
            Visible = true,
            Text = "PokéCorner",
            ContextMenuStrip = menu
        };
        _notifyIcon.DoubleClick += (_, _) => OpenCollectionRequested?.Invoke();
    }

        _notifyIcon.Icon = isActive ? _activeIcon : _inactiveIcon;

    }

    public void ShowNotification(string title, string text) =>
        _notifyIcon.ShowBalloonTip(timeout: 5000, tipTitle: title, tipText: text, tipIcon: ToolTipIcon.Info);

    public void Dispose()
    {
        _notifyIcon.Visible = false;
        _notifyIcon.Dispose();
    }
```

```csharp
using System.Runtime.InteropServices;
using System.Windows;
using System.Windows.Input;
using System.Windows.Interop;
using System.Windows.Media;

namespace PokeCorner.App.Overlay;

public enum ScreenCorner { TopLeft, TopRight, BottomLeft, BottomRight }

public partial class OverlayWindow : Window
{
    private const int GWL_EXSTYLE = -20;
    private const int WS_EX_TRANSPARENT = 0x00000020;
    private const int WS_EX_LAYERED = 0x00080000;

    public event Action? PetClicked;

    public OverlayWindow()
    {
        InitializeComponent();
        SourceInitialized += (_, _) => MakeClickThroughExceptSprite();
    }

    public void SetSpriteFrames(IReadOnlyList<ImageSource> frames)
    {
        SpriteImage.Source = frames.Count > 0 ? frames[0] : null;
    }

    public void PositionAt(ScreenCorner corner, int monitorIndex, int sizePixels)
    {
        Width = sizePixels;
        Height = sizePixels;

        var screens = System.Windows.Forms.Screen.AllScreens;
        var screen = monitorIndex >= 0 && monitorIndex < screens.Length ? screens[monitorIndex] : System.Windows.Forms.Screen.PrimaryScreen!;
        var bounds = screen.WorkingArea;

        const int margin = 12;
        (Left, Top) = corner switch
        {
            ScreenCorner.TopLeft => (bounds.Left + margin, bounds.Top + margin),
            ScreenCorner.TopRight => (bounds.Right - sizePixels - margin, bounds.Top + margin),
            ScreenCorner.BottomLeft => (bounds.Left + margin, bounds.Bottom - sizePixels - margin),
            ScreenCorner.BottomRight => (bounds.Right - sizePixels - margin, bounds.Bottom - sizePixels - margin),
            _ => (bounds.Right - sizePixels - margin, bounds.Bottom - sizePixels - margin)
        };
    }

    private void SpriteImage_MouseLeftButtonDown(object sender, MouseButtonEventArgs e)
    {
        if (e.ClickCount == 1)
        {
            PetClicked?.Invoke();
        }
        else if (e.ButtonState == MouseButtonState.Pressed)
        {
            DragMove();
        }
    }

    /// The window itself stays WS_EX_LAYERED | WS_EX_TRANSPARENT (click-through) at
    /// all times; the sprite Image re-enables hit-testing locally via IsHitTestVisible,
    /// so clicks land on the pet but pass through everywhere else in the window's bounds.
    private void MakeClickThroughExceptSprite()
    {
        var hwnd = new WindowInteropHelper(this).Handle;
        var extendedStyle = GetWindowLong(hwnd, GWL_EXSTYLE);
        SetWindowLong(hwnd, GWL_EXSTYLE, extendedStyle | WS_EX_LAYERED | WS_EX_TRANSPARENT);
        SpriteImage.IsHitTestVisible = true;
    }

    [DllImport("user32.dll")] private static extern int GetWindowLong(IntPtr hWnd, int nIndex);
    [DllImport("user32.dll")] private static extern int SetWindowLong(IntPtr hWnd, int nIndex, int dwNewLong);
}
```

> `WS_EX_TRANSPARENT` makes the *entire* window click-through at the Win32 level, which also swallows clicks on `SpriteImage` despite `IsHitTestVisible`. If manual testing (Step 3 below) shows the sprite isn't clickable, replace the whole-window transparent style with per-pixel hit testing instead: keep the window `WS_EX_LAYERED` only (drop `WS_EX_TRANSPARENT`), and override `HitTestCore` on a custom transparent-background element so only actual sprite pixels report hits — flag this as the first thing to check in manual verification.

```csharp
// src/PokeCorner.App/Popup/CollectionViewModel.cs
using PokeCorner.Core.Domain;

namespace PokeCorner.App.Popup;

public sealed record CollectionEntry(
    Guid IndividualId,
    int SpeciesId,
    bool IsShiny,
    bool IsActive,
    bool IsAvailableToActivate,
    DateTimeOffset? AvailableAgainAtUtc);

public sealed class CollectionViewModel
{
    private readonly PokemonCollection _collection;
    private readonly IDictionary<Guid, CareState> _careStates;
    private readonly CareModeService _careModeService;
    private readonly CareMode _careMode;

    public CollectionViewModel(PokemonCollection collection, IDictionary<Guid, CareState> careStates, CareModeService careModeService, CareMode careMode)
    {
        _collection = collection;
        _careStates = careStates;
        _careModeService = careModeService;
        _careMode = careMode;
    }

    public IReadOnlyList<CollectionEntry> GetEntries(DateTimeOffset nowUtc) =>
        _collection.Individuals.Select(individual =>
        {
            var careState = _careStates.TryGetValue(individual.Id, out var state) ? state : new CareState();
            var isAvailable = _careMode != CareMode.Hardcore || _careModeService.IsAvailableToPlaceOnDesktop(careState, nowUtc);
            return new CollectionEntry(
                individual.Id,
                individual.SpeciesId,
                individual.IsShiny,
                IsActive: individual.Id == _collection.ActiveIndividualId,
                IsAvailableToActivate: isAvailable,
                AvailableAgainAtUtc: isAvailable ? null : careState.RanAwayUntilUtc);
        }).ToList();

    public bool TrySetActive(Guid id, DateTimeOffset nowUtc)
    {
        var careState = _careStates.TryGetValue(id, out var state) ? state : new CareState();
        var isAvailable = _careMode != CareMode.Hardcore || _careModeService.IsAvailableToPlaceOnDesktop(careState, nowUtc);
        if (!isAvailable) return false;

        _collection.SetActive(id);
        return true;
    }

    public bool Release(Guid id) => _collection.Release(id);
}
```

```csharp
        _viewModel.Release(id);
```

```csharp
// src/PokeCorner.App/Autostart/AutostartRegistry.cs
using Microsoft.Win32;

namespace PokeCorner.App.Autostart;

public static class AutostartRegistry
{
    private const string RunKeyPath = @"Software\Microsoft\Windows\CurrentVersion\Run";
    private const string ValueName = "PokeCorner";

    public static void Enable(string executablePath)
    {
        using var key = Registry.CurrentUser.OpenSubKey(RunKeyPath, writable: true)!;
        key.SetValue(ValueName, $"\"{executablePath}\"");
    }

    public static void Disable()
    {
        using var key = Registry.CurrentUser.OpenSubKey(RunKeyPath, writable: true)!;
        if (key.GetValue(ValueName) is not null) key.DeleteValue(ValueName);
    }

    public static bool IsEnabled()
    {
        using var key = Registry.CurrentUser.OpenSubKey(RunKeyPath, writable: false);
        return key?.GetValue(ValueName) is not null;
    }
}
```

```csharp
using PokeCorner.Core.Domain;
using PokeCorner.Core.Persistence;

namespace PokeCorner.App.Onboarding;

public static class FirstRunFlow
{
    public static bool NeedsOnboarding(SaveData? loaded) =>
        loaded is null || loaded.Individuals.Count == 0;

    public static PokemonIndividual HatchStarter(CatchRoller roller, IReadOnlyList<SpeciesEvolutionChain> offlineStarters, DateTimeOffset nowUtc)
    {
        var result = roller.RollCatch(offlineStarters, ownedChainIds: new HashSet<int>(), ownedNaturesForDuplicateChain: new HashSet<Nature>());
        return new PokemonIndividual
        {
            SpeciesId = result.SpeciesId,
            EvolutionChainId = result.ChainId,
            ChainLength = result.ChainLength,
            Nature = result.Nature,
            IsShiny = result.IsShiny,
            CaughtAtUtc = nowUtc
        };
    }
}
```

```csharp
// src/PokeCorner.App/App.xaml.cs
using System.Windows;
using System.Windows.Threading;
using PokeCorner.App.Autostart;
using PokeCorner.App.Fullscreen;
using PokeCorner.App.Input;
using PokeCorner.App.Onboarding;
using PokeCorner.App.Overlay;
using PokeCorner.App.Popup;
using PokeCorner.App.Tray;
using PokeCorner.Core;
using PokeCorner.Core.ActivitySources;
using PokeCorner.Core.Domain;
using PokeCorner.Core.PokeApi;
using PokeCorner.Core.Persistence;

namespace PokeCorner.App;

public partial class App : Application
{
    private AppState _appState = null!;
    private JsonSaveStore _saveStore = null!;
    private SpriteCache _spriteCache = null!;
    private PokeApiClient _pokeApiClient = null!;
    private ClaudeCodeUsageProvider? _tokenUsageProvider;
    private GlobalInputHook? _inputHook;
    private InputEventCounter _inputCounter = null!;
    private DispatcherTimer _growthTickTimer = null!;
    private OverlayWindow _overlayWindow = null!;
    private TrayIconService _trayIconService = null!;
    private ForegroundWindowMonitor _fullscreenMonitor = null!;
    private CatchRoller _catchRoller = null!;
    private CareModeService _careModeService = null!;
    private FeedingService _feedingService = null!;
    private bool _isFullscreenHiding;

    protected override void OnStartup(StartupEventArgs e)
    {
        base.OnStartup(e);

        _saveStore = new JsonSaveStore(AppPaths.GetSaveFilePath());
        _spriteCache = new SpriteCache(AppPaths.GetSpriteCacheDirectory());
        _pokeApiClient = new PokeApiClient(new System.Net.Http.HttpClientHandler());
        _catchRoller = new CatchRoller();
        _careModeService = new CareModeService();
        _feedingService = new FeedingService();

        var loaded = _saveStore.Load();
        _appState = BuildAppState(loaded);

        if (FirstRunFlow.NeedsOnboarding(loaded))
        {
            var starter = FirstRunFlow.HatchStarter(_catchRoller, _spriteCache.LoadOfflineStarters(), DateTimeOffset.UtcNow);
            _appState.Collection.Add(starter);
        }

        _appState.CatchRollDue += OnCatchRollDue;

        _inputCounter = new InputEventCounter();
        if (_appState.Settings.ActivityTrackingEnabled)
        {
            _inputHook = new GlobalInputHook(_inputCounter);
            _inputHook.Start();
        }

        if (_appState.Settings.TokenTrackingEnabled)
        {
            _tokenUsageProvider = new ClaudeCodeUsageProvider(Environment.GetFolderPath(Environment.SpecialFolder.UserProfile) + "\\.claude");
        }

        _growthTickTimer = new DispatcherTimer { Interval = TimeSpan.FromSeconds(5) };
        _growthTickTimer.Tick += (_, _) => OnGrowthTick();
        _growthTickTimer.Start();

        _overlayWindow = new OverlayWindow();
        _overlayWindow.PositionAt(ParseCorner(_appState.Settings.ScreenCorner), _appState.Settings.MonitorIndex, _appState.Settings.PetSizePixels);
        _overlayWindow.PetClicked += TryFeedActiveIndividual;
        RefreshOverlayForActiveIndividual();

        _fullscreenMonitor = new ForegroundWindowMonitor(
            onFullscreenEntered: () => { _isFullscreenHiding = true; RefreshOverlayForActiveIndividual(); },
            onFullscreenExited: () => { _isFullscreenHiding = false; RefreshOverlayForActiveIndividual(); });
        if (_appState.Settings.AutoHideInFullscreen) _fullscreenMonitor.Start();

        _trayIconService = new TrayIconService();
        _trayIconService.SetActivityTrackingIndicator(_appState.Settings.ActivityTrackingEnabled);
        _trayIconService.OpenCollectionRequested += OpenCollectionWindow;
        _trayIconService.OpenSettingsRequested += OpenSettingsWindow;
        _trayIconService.ExitRequested += () => Shutdown();

        Exit += (_, _) => SaveState();
    }

    private AppState BuildAppState(SaveData? loaded)
    {
        var state = new AppState();
        if (loaded is not null)
        {
            state.Hydrate(loaded);
        }
        return state;
    }

    private void OnGrowthTick()
    {
        var units = _inputCounter.DrainUnits();
        if (_tokenUsageProvider is not null)
        {
            units += _tokenUsageProvider.ReadNewUnitsSinceLastCall();
        }
        if (units > 0)
        {
            _appState.ApplyActivityUnits(units, DateTimeOffset.UtcNow);
            _feedingService.EarnTreatsFromActivity(_appState.Feeding, units);
        }

        CheckHardcoreNeglect();
    }

    /// Hardcore care mode's "убегает" mechanic (spec: "Уход за питомцем") only has a
    /// visible effect if something polls for it — this is that poll, riding the same
    /// 5-second timer as growth so it doesn't need a timer of its own.
    private void CheckHardcoreNeglect()
    {
        if (_appState.Settings.CareMode != CareMode.Hardcore) return;

        var active = _appState.Collection.GetActive();
        if (active is null) return;

        var careState = _appState.GetOrCreateCareState(active.Id);
        var now = DateTimeOffset.UtcNow;
        if (!_careModeService.ShouldRunAway(careState, now, _appState.Settings.CareMode)) return;

        _careModeService.TriggerRunAway(careState, now);
        _appState.Collection.ClearActive();
        RefreshOverlayForActiveIndividual();
        _trayIconService.ShowNotification("PokéCorner", "Питомец расстроен и не хочет возвращаться на стол ещё час.");
    }

    /// Single entry point for "what should the overlay show right now" — called after
    /// startup, after switching/releasing in the collection popup, and after every
    /// fullscreen enter/exit, so the desktop sprite never drifts out of sync with
    /// AppState.Collection.GetActive().
    private void RefreshOverlayForActiveIndividual()
    {
        var active = _appState.Collection.GetActive();
        if (active is null)
        {
            _overlayWindow.Hide();
            return;
        }

        // Blocking on the sprite cache/download here is only ever slow on the very
        // first fetch of a given species (subsequent calls hit the on-disk cache from
        // SpriteCache); acceptable for now given how infrequently the active individual
        // changes, but convert to a fire-and-forget async load with a placeholder frame
        // if this is ever felt as UI jank.
        var spriteBytes = _spriteCache
            .GetOrFetchAsync(active.SpeciesId, () => _pokeApiClient.FetchSpriteAsync(active.SpeciesId))
            .GetAwaiter().GetResult();

        // GetOrFetchAsync returns null on a network failure (e.g. offline first launch
        // with nothing cached yet) rather than throwing — this is exactly the case the
        // spec's "работает без сети с первого раза" requirement covers. Skip the frame
        // update and leave whatever the overlay was already showing (or nothing, on a
        // true cold start) rather than crashing; RefreshOverlayForActiveIndividual will
        // be called again on the next state change or growth tick and can retry then.
        if (spriteBytes is not null)
        {
            // A single static frame for now — true Gen-V-style animated sprites come from
            // PokeAPI's separate animated-GIF sprite set, not the static front_default PNG
            // FetchSpriteAsync fetches; decoding multi-frame GIFs into FrameSequencer input
            // is Task 2 of docs/superpowers/plans/2026-08-19-pokecorner-full-experience-and-installer.md,
            // deliberately sequenced after this plan ships and is verified. SpriteAnimator/
            // FrameSequencer both already work correctly with a one-frame list (no visible
            // cycling, just a static sprite) so nothing here is broken in the meantime.
            _overlayWindow.SetSpriteFrames(new List<System.Windows.Media.ImageSource> { DecodeSprite(spriteBytes) });
        }

        if (_isFullscreenHiding)
        {
            _overlayWindow.Hide();
        }
        else
        {
            _overlayWindow.Show();
        }
    }

    private static System.Windows.Media.ImageSource DecodeSprite(byte[] bytes)
    {
        using var stream = new MemoryStream(bytes);
        var bitmap = new System.Windows.Media.Imaging.BitmapImage();
        bitmap.BeginInit();
        bitmap.CacheOption = System.Windows.Media.Imaging.BitmapCacheOption.OnLoad;
        bitmap.StreamSource = stream;
        bitmap.EndInit();
        bitmap.Freeze();
        return bitmap;
    }

    private void OnCatchRollDue()
    {
        var ownedChainIds = _appState.Collection.Individuals.Select(i => i.EvolutionChainId).ToHashSet();
        var availableChains = _spriteCache.LoadOfflineStarters(); // Widened to the full Gen 1-5 catalog by Task 1 of docs/superpowers/plans/2026-08-19-pokecorner-full-experience-and-installer.md, once this plan ships and is verified.
        var ownedNatures = _appState.Collection.Individuals
            .Where(i => ownedChainIds.Contains(i.EvolutionChainId))
            .Select(i => i.Nature)
            .ToHashSet();

        var result = _catchRoller.RollCatch(availableChains, ownedChainIds, ownedNatures);
        _appState.Collection.Add(new PokemonIndividual
        {
            SpeciesId = result.SpeciesId,
            EvolutionChainId = result.ChainId,
            ChainLength = result.ChainLength,
            Nature = result.Nature,
            IsShiny = result.IsShiny,
            CaughtAtUtc = DateTimeOffset.UtcNow
        });
    }

    private void TryFeedActiveIndividual()
    {
        var active = _appState.Collection.GetActive();
        if (active is null) return;

        var careState = _appState.GetOrCreateCareState(active.Id);
        _feedingService.TryFeed(_appState.Settings.FeedingMode, _appState.Feeding, careState, DateTimeOffset.UtcNow);
    }

    private void OpenCollectionWindow()
    {
        var vm = new CollectionViewModel(_appState.Collection, _appState.CareStateByIndividualId, _careModeService, _appState.Settings.CareMode);
        new Window { Content = new CollectionView(vm, _appState, TryFeedActiveIndividual, RefreshOverlayForActiveIndividual), Title = "Коллекция", Width = 420, Height = 480 }.Show();
    }

    private void OpenSettingsWindow()
    {
        var vm = new SettingsViewModel(_appState.Settings);
        new Window { Content = new SettingsView(vm), Title = "Настройки", Width = 360, Height = 480 }.Show();
    }

    private static ScreenCorner ParseCorner(string corner) => corner switch
    {
        "TopLeft" => ScreenCorner.TopLeft,
        "TopRight" => ScreenCorner.TopRight,
        "BottomLeft" => ScreenCorner.BottomLeft,
        _ => ScreenCorner.BottomRight
    };

    private void SaveState()
    {
        _inputHook?.Dispose();
        _fullscreenMonitor?.Dispose();
        _trayIconService?.Dispose();

        var data = new SaveData
        {
            Individuals = _appState.Collection.Individuals.ToList(),
            ActiveIndividualId = _appState.Collection.ActiveIndividualId,
            Settings = _appState.Settings,
            FeedingTreats = _appState.Feeding.Treats,
            CareStates = new Dictionary<Guid, CareState>(_appState.CareStateByIndividualId)
        };
        _saveStore.Save(data);
    }
}
```
